        // --- 3. WEB AUDIO API ---
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        function playTone(type) {
            if(audioCtx.state === 'suspended') audioCtx.resume();
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            
            if(type === 'correct') {
                osc.type = 'sine';
                osc.frequency.setValueAtTime(880, audioCtx.currentTime); // A5
                osc.frequency.exponentialRampToValueAtTime(1760, audioCtx.currentTime + 0.1);
                gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.1);
                osc.start(); osc.stop(audioCtx.currentTime + 0.1);
            } else {
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(200, audioCtx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(100, audioCtx.currentTime + 0.2);
                gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.2);
                osc.start(); osc.stop(audioCtx.currentTime + 0.2);
            }
        }

        // --- 4 & 5. APP CONTROLLER & DRAG LOGIC ---
        const app = {
            studentInfo: {},
            currentGameData: null,
            startTime: 0,
            timerInterval: null,
            matchedCount: 0,
            answers: {}, 

            login() {
                const name = document.getElementById('student-name').value.trim();
                const cls = document.getElementById('student-class').value.trim();
                if(!name || !cls) return alert("Vui lòng nhập đầy đủ Họ tên và Lớp!");
                this.studentInfo = { name, cls };
                document.getElementById('display-name').innerText = name;
                document.getElementById('login-view').classList.add('hidden');
                this.renderMenu();
                this.showMenu();
            },

            renderMenu() {
                const grid = document.getElementById('game-list');
                grid.innerHTML = '';
                gameData.forEach((g, idx) => {
                    const card = document.createElement('div');
                    card.className = 'game-card';
                    card.innerHTML = `<h4>${g.title}</h4><p style="font-size:12px; margin-top:10px;">Gồm ${g.targets.length} câu hỏi</p>`;
                    card.onclick = () => this.startGame(idx);
                    grid.appendChild(card);
                });
            },

            showMenu() {
                document.querySelectorAll('.container > div').forEach(d => d.classList.add('hidden'));
                document.getElementById('menu-view').classList.remove('hidden');
                clearInterval(this.timerInterval);
            },

            startGame(idx) {
                this.currentGameData = gameData[idx];
                this.matchedCount = 0;
                this.answers = {};
                
                document.getElementById('menu-view').classList.add('hidden');
                document.getElementById('report-view').classList.add('hidden');
                document.getElementById('game-view').classList.remove('hidden');
                
                document.getElementById('game-title').innerText = this.currentGameData.title;
                document.getElementById('game-instruction').innerText = this.currentGameData.instruction;
                document.getElementById('btn-finish').classList.add('hidden');

                // Render Targets
                const dropContainer = document.getElementById('drop-container');
                dropContainer.innerHTML = '';
                this.currentGameData.targets.forEach(t => {
                    const dz = document.createElement('div');
                    dz.className = 'dropzone';
                    dz.dataset.id = t.id;
                    dz.dataset.label = t.label;
                    dropContainer.appendChild(dz);
                });

                // Render Draggables (Shuffle)
                const dragContainer = document.getElementById('drag-container');
                dragContainer.innerHTML = '';
                let drags = [...this.currentGameData.drags].sort(() => Math.random() - 0.5);
                
                drags.forEach(d => {
                    const el = document.createElement('div');
                    el.className = 'draggable';
                    el.innerText = d.content;
                    el.dataset.id = d.id;
                    el.dataset.match = d.match || "none";
                    this.initDrag(el);
                    dragContainer.appendChild(el);
                });

                this.startTimer();
            },

            startTimer() {
                this.startTime = Date.now();
                const timerEl = document.getElementById('game-timer');
                this.timerInterval = setInterval(() => {
                    const diff = Math.floor((Date.now() - this.startTime) / 1000);
                    const m = String(Math.floor(diff / 60)).padStart(2, '0');
                    const s = String(diff % 60).padStart(2, '0');
                    timerEl.innerText = `${m}:${s}`;
                }, 1000);
            },

            initDrag(el) {
                let startX, startY, initialX, initialY, clonePlaceholder;

                const startDrag = (e) => {
                    if (el.classList.contains('dropped')) return;
                    e.preventDefault(); // Ngăn cuộn trang trên mobile
                    const touch = e.type === 'touchstart' ? e.touches[0] : e;
                    
                    const rect = el.getBoundingClientRect();
                    startX = touch.clientX;
                    startY = touch.clientY;
                    
                    // Tạo placeholder giữ vị trí để không hỏng layout
                    clonePlaceholder = el.cloneNode(true);
                    clonePlaceholder.style.visibility = 'hidden';
                    clonePlaceholder.style.opacity = '0';
                    el.parentNode.insertBefore(clonePlaceholder, el);
                    
                    // Set absolute
                    el.style.position = 'fixed';
                    el.style.left = rect.left + 'px';
                    el.style.top = rect.top + 'px';
                    el.style.width = rect.width + 'px';
                    initialX = rect.left;
                    initialY = rect.top;
                    
                    el.classList.add('dragging');
                    el.style.transition = 'none'; // Bỏ transition khi đang kéo

                    document.addEventListener(e.type === 'touchstart' ? 'touchmove' : 'mousemove', moveDrag, {passive: false});
                    document.addEventListener(e.type === 'touchstart' ? 'touchend' : 'mouseup', endDrag);
                };

                const moveDrag = (e) => {
                    e.preventDefault();
                    const touch = e.type === 'touchmove' ? e.touches[0] : e;
                    const dx = touch.clientX - startX;
                    const dy = touch.clientY - startY;
                    el.style.transform = `translate(${dx}px, ${dy}px)`;
                };

                const endDrag = (e) => {
                    document.removeEventListener(e.type === 'touchend' ? 'touchmove' : 'mousemove', moveDrag);
                    document.removeEventListener(e.type === 'touchend' ? 'touchend' : 'mouseup', endDrag);
                    
                    el.classList.remove('dragging');
                    const touch = e.type === 'touchend' ? e.changedTouches[0] : e;
                    
                    // Ẩn tạm phần tử kéo để lấy phần tử bên dưới (Dropzone)
                    el.style.display = 'none';
                    const targetEl = document.elementFromPoint(touch.clientX, touch.clientY);
                    el.style.display = 'block';

                    const dropzone = targetEl ? targetEl.closest('.dropzone') : null;

                    if (dropzone && dropzone.dataset.id === el.dataset.match && !dropzone.classList.contains('correct')) {
                        // Thả ĐÚNG
                        playTone('correct');
                        dropzone.appendChild(el);
                        dropzone.classList.add('correct');
                        el.style.position = 'static';
                        el.style.transform = 'none';
                        el.style.width = 'auto';
                        el.classList.add('dropped');
                        
                        const label = document.createElement('span');
                        label.className = 'correct-label';
                        label.innerText = 'Chính xác!';
                        dropzone.appendChild(label);
                        
                        clonePlaceholder.remove();
                        
                        // Lưu đáp án
                        this.answers[dropzone.dataset.id] = true;
                        this.matchedCount++;
                        
                        if(this.matchedCount === this.currentGameData.targets.length) {
                            document.getElementById('btn-finish').classList.remove('hidden');
                        }
                    } else {
                        // Thả SAI -> Bounce back
                        playTone('wrong');
                        el.style.transition = 'transform 0.4s cubic-bezier(0.175, 0.885, 0.32, 1.275)';
                        el.style.transform = 'translate(0, 0)';
                        
                        setTimeout(() => {
                            el.style.position = 'static';
                            el.style.transition = 'none';
                            clonePlaceholder.parentNode.replaceChild(el, clonePlaceholder);
                        }, 400);
                    }
                };

                el.addEventListener('mousedown', startDrag);
                el.addEventListener('touchstart', startDrag, {passive: false});
            },

            // --- 6. SCORING & REPORTING ---
            finishGame() {
                clearInterval(this.timerInterval);
                const timeStr = document.getElementById('game-timer').innerText;
                const totalTargets = this.currentGameData.targets.length;
                
                // Mặc định do bắt buộc kéo thả đúng mới cho xong nên điểm là 100%. 
                // Ở phiên bản này học sinh sẽ học qua việc thử lỗi. 
                const scorePercent = 100; 

                // Lưu lịch sử
                const history = JSON.parse(safeStorage.getItem('gameHistory') || '[]');
                history.push({
                    name: this.studentInfo.name,
                    cls: this.studentInfo.cls,
                    game: this.currentGameData.title,
                    score: scorePercent,
                    time: timeStr,
                    date: new Date().toLocaleDateString('vi-VN')
                });
                safeStorage.setItem('gameHistory', JSON.stringify(history));

                // Chuyển view
                document.getElementById('game-view').classList.add('hidden');
                document.getElementById('report-view').classList.remove('hidden');

                // Render Progress
                setTimeout(() => {
                    document.getElementById('score-radial').style.setProperty('--progress', `${scorePercent}%`);
                    document.getElementById('score-text').innerText = `${scorePercent}%`;
                }, 100);

                // Feedback
                document.getElementById('grade-text').innerText = "Xuất sắc!";
                document.getElementById('grade-text').style.color = "var(--success)";
                document.getElementById('feedback-text').innerText = "Bạn đã nắm rất vững kiến thức về Hình bình hành. Hãy phát huy nhé!";

                // Render Review Table
                const tbody = document.querySelector('#review-table tbody');
                tbody.innerHTML = '';
                this.currentGameData.targets.forEach(t => {
                    const dragObj = this.currentGameData.drags.find(d => d.match === t.id);
                    const tr = document.createElement('tr');
                    tr.innerHTML = `
                        <td>${t.label}</td>
                        <td style="color:var(--success); font-weight:bold;">${dragObj.content}</td>
                        <td>${dragObj.exp}</td>
                    `;
                    tbody.appendChild(tr);
                });
            },

            // --- 7. HISTORY & EXPORT ---
            showHistory() {
                document.querySelectorAll('.container > div').forEach(d => d.classList.add('hidden'));
                document.getElementById('history-view').classList.remove('hidden');
                
                const history = JSON.parse(safeStorage.getItem('gameHistory') || '[]');
                const tbody = document.querySelector('#history-table tbody');
                tbody.innerHTML = '';
                if(history.length === 0) {
                    tbody.innerHTML = '<tr><td colspan="6">Chưa có dữ liệu lịch sử</td></tr>';
                    return;
                }
                history.forEach(h => {
                    const tr = document.createElement('tr');
                    tr.innerHTML = `<td>${h.name}</td><td>${h.cls}</td><td>${h.game}</td><td>${h.score}%</td><td>${h.time}</td><td>${h.date}</td>`;
                    tbody.appendChild(tr);
                });
            },

            exportCSV() {
                const history = JSON.parse(safeStorage.getItem('gameHistory') || '[]');
                if(history.length === 0) return alert("Không có dữ liệu để xuất!");
                
                let csvContent = '\uFEFFHọ Tên;Lớp;Trò Chơi;Điểm Số;Thời Gian;Ngày Chơi\n';
                history.forEach(h => {
                    csvContent += `${h.name};${h.cls};${h.game};${h.score}%;${h.time};${h.date}\n`;
                });

                const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
                const link = document.createElement("a");
                link.href = URL.createObjectURL(blob);
                link.download = `BangDiem_HinhBinhHanh_${new Date().getTime()}.csv`;
                link.click();
            },

            exportPDF() {
                window.print();
            }
        };
